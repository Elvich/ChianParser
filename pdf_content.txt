4.  Robust  Upper-Market  Percentile  Benchmarking  
To  measure  the  true  arbitrage  potential  against  target  renovated  exit  prices  instead  of  the  
depressed
 
local
 
median
 
baseline,
 
the
 
scoring
 
analyzer
 
must
 
implement
 
a
 
secure
 
statistical
 
percentile
 
selector.
 
struct  MarketBenchmarkAnalyzer  {  
    
static
 
func
 
calculateTargetExitPrice(from
 
apartments:
 
[Apartment],
 
config:
 
ScoringConfiguration)
 
->
 
Double?
 
{
 
        
guard
 
!apartments.isEmpty
 
else
 
{
 
return
 
nil
 
}
 
        
 
        
let
 
sortedPricesSqm
 
=
 
apartments.map
 
{
 
$0.priceSqm
 
}.sorted()
 
        
 
        
//
 
Toggle
 
routing
 
logic
 
        
let
 
targetPercentile
 
=
 
config.isPercentileBenchmarkEnabled
 
?
 
config.targetPercentile
 
:
 
0.50
 
        
 
        
let
 
targetIndex
 
=
 
Int(Double(sortedPricesSqm.count)
 
*
 
targetPercentile)
 
        
let
 
safeIndex
 
=
 
max(0,
 
min(targetIndex,
 
sortedPricesSqm.count
 
-
 
1))
 
        
 
        
return
 
sortedPricesSqm[safeIndex]
 
    
}
 
}
